class PortDetectionApp:
    def __init__(self, root):
        self.root = root
        self.root.title("Port Detection Dashboard")

        self.detection_running = False
        self.splitter_results = defaultdict(list)
        self.port_status_count = defaultdict(int)
        self.ocr_cache = {}
        self.frame_count = 0
        self.current_splitter_num = None  # Tracks the current splitter

        self.conf_threshold = 0.4
        self.feed_width = 640
        self.feed_height = 480

        self.setup_ui()
        self.setup_model()

    def setup_ui(self):
        self.create_control_frame()
        self.create_counters_frame()
        self.create_splitter_frame()
        self.create_feed_frame()

    def create_control_frame(self):
        control_frame = tk.Frame(self.root, bg="#F4EFE6")
        control_frame.grid(row=0, column=0, columnspan=5, pady=10, padx=10, sticky="ew")

        buttons = [
            ("Start Detection", self.start_detection, "#2ECC71"),
            ("Stop Detection", self.stop_detection, "#E74C3C"),
            ("Retry Last Splitter", self.retry_last_splitter, "#F1C40F"),
            ("Load Video File", self.load_video, "#9B59B6"),
            ("Start Video Detection", self.start_video_detection, "#1ABC9C")
        ]

        for i, (text, command, color) in enumerate(buttons):
            Button(control_frame, text=text, command=command, bg=color, fg="white", 
                   font=("Helvetica", 12, "bold")).grid(row=0, column=i, padx=5)

    def create_counters_frame(self):
        self.counters_frame = tk.Frame(self.root, bg="#E9DFCE")
        self.counters_frame.grid(row=1, column=0, columnspan=5, pady=10, padx=10, sticky="ew")

        self.counter_labels = {}
        counters = ["Detected Splitters", "Completed Splitters", "Detected Ports", 
                    "Connected Ports", "Capped Ports", "Uncapped Ports"]

        for i, counter in enumerate(counters):
            label = tk.Label(self.counters_frame, text=f"{counter}: 0", bg="#E9DFCE",
                             fg="#1C160C", font=("Helvetica", 12))
            label.grid(row=i//3, column=i%3, padx=10, sticky="w")
            self.counter_labels[counter] = label

    def create_splitter_frame(self):
        splitter_frame = tk.Frame(self.root, bg="#FFFFFF")
        splitter_frame.grid(row=2, column=0, columnspan=2, padx=10, pady=10)

        self.splitter_frames = []
        for i in range(4):
            frame = tk.Frame(splitter_frame, bg="#F4EFE6", pady=5)
            frame.grid(row=i, column=0, pady=5)
            tk.Label(frame, text=f"Splitter {i + 1}", bg="#F4EFE6", fg="#1C160C", 
                     font=("Helvetica", 12)).grid(row=0, column=0, columnspan=8)
            
            for j in range(16):
                btn = tk.Button(frame, text=str(j + 1), width=5, height=2, bg="#95A5A6", 
                                fg="white", font=("Helvetica", 10, "bold"))
                btn.grid(row=1 + (j // 8), column=j % 8, padx=2, pady=2)
            self.splitter_frames.append(frame)

    def create_feed_frame(self):
        self.feed_frame = tk.Frame(self.root, bg="#FFFFFF")
        self.feed_frame.grid(row=2, column=2, columnspan=3, padx=10, pady=10)
        self.feed_label = Label(self.feed_frame)
        self.feed_label.grid(row=0, column=0)

    def setup_model(self):
        self.device = 'cuda' if torch.cuda.is_available() else 'cpu'
        logging.info(f"Using device: {self.device}")
        # self.model = YOLO(r'C:\Users\007\Desktop\progettov2\best.pt').to(self.device)
        logging.info(f"YOLOv8 model loaded on {self.device}")
        self.cap = None

    def start_detection(self):
        self.cap = cv2.VideoCapture(0)
        if not self.cap.isOpened():
            logging.error("Cannot open camera")
            return
        self.detection_running = True
        self.splitter_results.clear()
        self.frame_count = 0
        self.update_feed()

    def stop_detection(self):
        self.detection_running = False
        if self.cap:
            self.cap.release()
        self.feed_label.configure(image='')
        logging.info("Detection stopped.")

    def retry_last_splitter(self):
        if self.splitter_results:
            last_splitter = max(self.splitter_results.keys())
            self.update_splitter_buttons(last_splitter, self.splitter_results[last_splitter])
        else:
            logging.warning("No previous results available for retry")

    def load_video(self):
        video_path = filedialog.askopenfilename(title="Select a video file", 
                                                filetypes=[("Video Files", "*.mp4 *.avi *.mov *.mkv *.flv")])
        if video_path:
            self.cap = cv2.VideoCapture(video_path)
            if not self.cap.isOpened():
                logging.error(f"Cannot open video file: {video_path}")
                return
            self.splitter_results.clear()
            self.frame_count = 0
            self.start_video_detection()

    def start_video_detection(self):
        if self.cap and self.cap.isOpened():
            self.detection_running = True
            self.update_feed()
        else:
            logging.error("No video loaded or video cannot be opened properly")

    def update_feed(self):
        if not self.detection_running or not self.cap:

        ret, frame = self.cap.read()
        if not ret:
            self.stop_detection()
            return

        # Resize the frame to ensure consistency
        frame = cv2.resize(frame, (self.feed_width, self.feed_height))

        self.frame_count += 1

    def process_detections(self, frame, detections):
        annotated_frame = frame.copy()
        splitter_num = None  # Variable to store the detected splitter

        for box in detections.boxes:
            conf = float(box.conf)
            cls = int(box.cls)
            if conf >= self.conf_threshold:
                self.annotate_frame(annotated_frame, box, cls, conf)

                # If a barcode or splitter is detected, update the splitter number
                if cls == 0:  # Assume class 0 is for barcodes
                    splitter_num = self.decode_barcode_or_ocr(annotated_frame, box)

        # Change splitter only when a new barcode is detected
        if splitter_num and splitter_num != self.current_splitter_num:
            logging.info(f"New splitter detected: {splitter_num}")
            self.current_splitter_num = splitter_num
            self.splitter_results[self.current_splitter_num] = []

        # Add detected ports to the current splitter
        if splitter_num:
            self.splitter_results[self.current_splitter_num].append((cls, box))

        self.update_splitter_buttons(self.current_splitter_num, self.splitter_results[self.current_splitter_num])
        self.update_counter_labels()
        self.display_frame(annotated_frame)

    def decode_barcode_or_ocr(self, frame, box):
        x1, y1, x2, y2 = map(int, box.xyxy[0].cpu().numpy())
        label_area = frame[y1:y2, x1:x2]

        # Decode the barcode
        decoded_objects = pyzbar.decode(label_area)
        for obj in decoded_objects:
            splitter_num = obj.data.decode('utf-8')
            logging.info(f"Decoded barcode: {splitter_num}")
            return splitter_num

        # If the barcode is not detected, use OCR
        gray_label_area = cv2.cvtColor(label_area, cv2.COLOR_BGR2GRAY)
        _, thresh_label_area = cv2.threshold(gray_label_area, 150, 255, cv2.THRESH_BINARY)
        ocr_result = pytesseract.image_to_string(thresh_label_area).strip()

        logging.debug(f"OCR Result: {ocr_result}")
        return ocr_result

    def annotate_frame(self, frame, box, cls, conf):
        labels = ['barcode', 'connected port', 'port with green cap', 'port without cap', 'splitter']
        colors = [(0, 0, 0),   # black for barcode
                  (0, 255, 0), # green for connected port
                  (255, 255, 0), # yellow for port with green cap
                  (255, 0, 0),   # red for port without cap
                  (0, 0, 255)]   # blue for splitter
        
        label = labels[cls]
        color = colors[cls]
        x1, y1, x2, y2 = map(int, box.xyxy[0].cpu().numpy())

        cv2.rectangle(frame, (x1, y1), (x2, y2), color, 2)
        cv2.putText(frame, f'{label} ({conf:.2f})', (x1, y1 - 10), cv2.FONT_HERSHEY_SIMPLEX, 0.5, color, 2)

    def update_splitter_buttons(self, splitter_index, detections):
        colors = ["blue", "green", "yellow", "red", "cyan"]
        for btn in self.splitter_frames[splitter_index % 4].winfo_children():
            if isinstance(btn, tk.Button):
                btn.configure(bg="#95A5A6")  # Reset to default color

        for cls, box in detections:
            color = colors[cls]
            x_center = (box.xyxy[0][0].item() + box.xyxy[0][2].item()) / 2
            port_index = int(x_center // (self.feed_width / 16))  # Based on feed video width

            if 0 <= port_index < 16:
                btn = self.splitter_frames[splitter_index % 4].winfo_children()[port_index + 1]
                btn.configure(bg=color)

    def update_counter_labels(self):
        counts = defaultdict(int)
        for detections in self.splitter_results.values():
            for cls, _ in detections:
                if cls == 4:  # splitter
                    counts["Detected Splitters"] += 1
                else:
                    counts["Detected Ports"] += 1
                    if cls == 1:
                        counts["Connected Ports"] += 1
                    elif cls == 2:
                        counts["Capped Ports"] += 1
                    elif cls == 3:
                        counts["Uncapped Ports"] += 1

        counts["Completed Splitters"] = len(self.splitter_results)

        for counter, value in counts.items():
            self.counter_labels[counter].config(text=f"{counter}: {value}")

    def display_frame(self, frame):
        frame_rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
        img = Image.fromarray(frame_rgb)
        imgtk = ImageTk.PhotoImage(image=img)
        self.feed_label.imgtk = imgtk
        self.feed_label.configure(image=imgtk)

    def __del__(self):
        if self.cap:
            self.cap.release()

if __name__ == "__main__":
    root = tk.Tk()
